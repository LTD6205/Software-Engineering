
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. US

CREATE TABLE Users (
    user_id     UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name        VARCHAR(100)        NOT NULL,
    email       VARCHAR(255)        NOT NULL UNIQUE,
    role        VARCHAR(20)         NOT NULL DEFAULT 'staff'
                    CHECK (role IN ('manager', 'staff', 'admin')),
    created_at  TIMESTAMP           NOT NULL DEFAULT NOW()
);


--  2. EVENTS

CREATE TABLE Events (
    event_id    UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    event_name  VARCHAR(255)        NOT NULL,
    description TEXT,
    start_time  TIMESTAMP           NOT NULL,
    end_time    TIMESTAMP           NOT NULL,
    created_by  UUID                NOT NULL REFERENCES Users(user_id),
    status      VARCHAR(20)         NOT NULL DEFAULT 'pending'
                    CHECK (status IN ('pending', 'in_progress', 'completed')),
    created_at  TIMESTAMP           NOT NULL DEFAULT NOW(),

    CONSTRAINT events_time_check CHECK (end_time > start_time)
);


--  3. AI_REQUESTS
--  Must be defined BEFORE Task_Logs (Task_Logs references it)

CREATE TABLE AI_Requests (
    request_id  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id     UUID                NOT NULL REFERENCES Users(user_id),
    prompt      TEXT                NOT NULL,
    response    JSONB,                          -- structured JSON output from AI
    status      VARCHAR(20)         NOT NULL DEFAULT 'pending'
                    -- AI confirm/clarification flow adds awaiting_confirmation, cancelled, needs_clarification
                    CHECK (status IN (
                        'pending', 'success', 'rejected',
                        'awaiting_confirmation', 'cancelled', 'needs_clarification'
                    )),
    created_at  TIMESTAMP           NOT NULL DEFAULT NOW()
);


--  4. TASKS
CREATE TABLE Tasks (
    task_id         UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    event_id        UUID                NOT NULL REFERENCES Events(event_id) ON DELETE CASCADE,
    task_name       VARCHAR(255)        NOT NULL,
    description     TEXT,
    priority_label  VARCHAR(10)         NOT NULL DEFAULT 'medium'
                        CHECK (priority_label IN ('low', 'medium', 'high')),
    priority_score  INT                 NOT NULL DEFAULT 0,  -- calculated & sorted by backend
    priority_source VARCHAR(10)         NOT NULL DEFAULT 'user'
                        CHECK (priority_source IN ('user', 'ai', 'auto')),
    -- A new task starts In Progress (there is no "pending" task stage); the
    -- 'pending' value is kept in the CHECK only for rows created before this
    -- policy. See migrations/2026-06-02_drop_pending_status.sql.
    status          VARCHAR(20)         NOT NULL DEFAULT 'in_progress'
                        CHECK (status IN ('pending', 'in_progress', 'completed', 'overdue')),
    start_time      TIMESTAMP,
    deadline        TIMESTAMP,
    created_by      UUID                NOT NULL REFERENCES Users(user_id),
    created_at      TIMESTAMP           NOT NULL DEFAULT NOW(),

    CONSTRAINT tasks_time_check CHECK (deadline IS NULL OR start_time IS NULL OR deadline > start_time)
);


--  5. TASK_LOGS
--  Tracks every change to a task (by a user OR by AI)
CREATE TABLE Task_Logs (
    log_id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    task_id         UUID                NOT NULL REFERENCES Tasks(task_id) ON DELETE CASCADE,
    action_type     VARCHAR(50)         NOT NULL,   -- e.g. 'status_change', 'reassign', 'deadline_update'
    old_value       JSONB,
    new_value       JSONB,
    actor_type      VARCHAR(10)         NOT NULL
                        CHECK (actor_type IN ('user', 'ai', 'system')),
    actor_user_id   UUID                REFERENCES Users(user_id),      -- only when actor_type = 'user'
    ai_request_id   UUID                REFERENCES AI_Requests(request_id), -- only when actor_type = 'ai'
    created_at      TIMESTAMP           NOT NULL DEFAULT NOW(),

    -- Enforce the right actor id for each actor_type. 'system' (e.g. the overdue
    -- cron) carries neither a user nor an AI-request id.
    CONSTRAINT task_logs_actor_check CHECK (
        (actor_type = 'user'   AND actor_user_id IS NOT NULL AND ai_request_id IS NULL) OR
        (actor_type = 'ai'     AND ai_request_id IS NOT NULL AND actor_user_id IS NULL) OR
        (actor_type = 'system' AND actor_user_id IS NULL     AND ai_request_id IS NULL)
    )
);


--  6. TASK_ASSIGNMENTS
--  Links one or more users to a task
CREATE TABLE Task_Assignments (
    assignment_id   UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    task_id         UUID                NOT NULL REFERENCES Tasks(task_id) ON DELETE CASCADE,
    user_id         UUID                NOT NULL REFERENCES Users(user_id),
    assigned_at     TIMESTAMP           NOT NULL DEFAULT NOW(),

    CONSTRAINT task_assignments_unique UNIQUE (task_id, user_id)
);


--  7. TASK_DEPENDENCIES
--  "task_id cannot start until depends_on_task is completed"
CREATE TABLE Task_Dependencies (
    dependency_id   UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    task_id         UUID                NOT NULL REFERENCES Tasks(task_id) ON DELETE CASCADE,
    depends_on_task UUID                NOT NULL REFERENCES Tasks(task_id) ON DELETE CASCADE,

    CONSTRAINT task_dependencies_no_self CHECK (task_id <> depends_on_task),
    CONSTRAINT task_dependencies_unique  UNIQUE (task_id, depends_on_task)
);


--  8. NOTIFICATIONS
CREATE TABLE Notifications (
    notification_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id         UUID                NOT NULL REFERENCES Users(user_id),
    task_id         UUID                REFERENCES Tasks(task_id) ON DELETE CASCADE,
    event_id        UUID                REFERENCES Events(event_id) ON DELETE CASCADE,
    type            VARCHAR(20)         NOT NULL
                        CHECK (type IN ('reminder', 'alert', 'overdue',
                                        'event', 'task', 'reassignment', 'info')),
    message         TEXT                NOT NULL,
    is_read         BOOLEAN             NOT NULL DEFAULT FALSE,
    created_at      TIMESTAMP           NOT NULL DEFAULT NOW()
);


--  9. AI_TASK_MAP
--  Links an AI request to the tasks it created or modified
CREATE TABLE AI_Task_Map (
    id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    request_id  UUID    NOT NULL REFERENCES AI_Requests(request_id),
    task_id     UUID    NOT NULL REFERENCES Tasks(task_id) ON DELETE CASCADE,

    CONSTRAINT ai_task_map_unique UNIQUE (request_id, task_id)
);


--  INDEXES (speeds up common queries)

-- Look up tasks by event
CREATE INDEX idx_tasks_event_id      ON Tasks(event_id);

-- Look up tasks by status (e.g. find all overdue)
CREATE INDEX idx_tasks_status        ON Tasks(status);

-- Look up tasks by deadline (used by cron job for notifications)
CREATE INDEX idx_tasks_deadline      ON Tasks(deadline);

-- Look up logs by task
CREATE INDEX idx_task_logs_task_id   ON Task_Logs(task_id);

-- Look up assignments by user or task
CREATE INDEX idx_assignments_user    ON Task_Assignments(user_id);
CREATE INDEX idx_assignments_task    ON Task_Assignments(task_id);

-- Look up unread notifications per user
CREATE INDEX idx_notif_user_unread   ON Notifications(user_id, is_read);

-- Look up dependencies by task
CREATE INDEX idx_deps_task_id        ON Task_Dependencies(task_id);

    CHECK (role IN ('manager', 'staff', 'admin', 'organizer'));

-- 6. Which managers are part of an event (their staff are members too).
CREATE TABLE IF NOT EXISTS event_managers (
    event_id   UUID NOT NULL REFERENCES events(event_id) ON DELETE CASCADE,
    manager_id UUID NOT NULL REFERENCES users(user_id),
    PRIMARY KEY (event_id, manager_id)
);
CREATE INDEX IF NOT EXISTS idx_event_managers_manager ON event_managers(manager_id);

-- 7. Make task/event child foreign keys ON DELETE CASCADE, so deleting a task
--    or event can never be blocked by (or orphan) its child rows. The CREATE
--    TABLE statements above already include this for fresh installs; this block
--    upgrades an EXISTING database. Idempotent — safe to re-run.
--    (Same SQL as event-ops-backend/migrations/2026-06-02_fk_on_delete_cascade.sql;
--     run it with `npm run db:migrate` or paste this block into pgAdmin.)
CREATE OR REPLACE FUNCTION pg_temp.recreate_fk_cascade(
    p_table text, p_col text, p_reftable text, p_refcol text
) RETURNS void AS $$
DECLARE c text;
BEGIN
    FOR c IN
        SELECT con.conname
        FROM pg_constraint con
        JOIN pg_class      rel ON rel.oid = con.conrelid
        JOIN pg_namespace  ns  ON ns.oid  = rel.relnamespace
        JOIN pg_attribute  att ON att.attrelid = rel.oid AND att.attnum = ANY (con.conkey)
        WHERE con.contype = 'f' AND ns.nspname = 'public'
          AND rel.relname = p_table AND att.attname = p_col
    LOOP
        EXECUTE format('ALTER TABLE %I DROP CONSTRAINT %I', p_table, c);
    END LOOP;
    EXECUTE format(
        'ALTER TABLE %I ADD CONSTRAINT %I FOREIGN KEY (%I) REFERENCES %I(%I) ON DELETE CASCADE',
        p_table, p_table || '_' || p_col || '_fkey', p_col, p_reftable, p_refcol
    );
END;
$$ LANGUAGE plpgsql;

SELECT pg_temp.recreate_fk_cascade('task_logs',         'task_id',         'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('task_assignments',  'task_id',         'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('task_dependencies', 'task_id',         'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('task_dependencies', 'depends_on_task', 'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('ai_task_map',       'task_id',         'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('notifications',     'task_id',         'tasks',  'task_id');
SELECT pg_temp.recreate_fk_cascade('tasks',             'event_id',        'events', 'event_id');

-- 8. Merged tasks: a "task group" is a named parent several tasks belong to.
--    (Same SQL as event-ops-backend/migrations/2026-06-02_task_groups.sql.)
CREATE TABLE IF NOT EXISTS task_groups (
    group_id   UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    event_id   UUID         NOT NULL REFERENCES events(event_id) ON DELETE CASCADE,
    title      VARCHAR(255) NOT NULL DEFAULT '',
    created_at TIMESTAMP    NOT NULL DEFAULT NOW()
);
ALTER TABLE tasks ADD COLUMN IF NOT EXISTS group_id UUID;
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'tasks_group_id_fkey') THEN
        ALTER TABLE tasks ADD CONSTRAINT tasks_group_id_fkey
            FOREIGN KEY (group_id) REFERENCES task_groups(group_id) ON DELETE SET NULL;
    END IF;
END $$;
CREATE INDEX IF NOT EXISTS idx_tasks_group ON tasks(group_id);

-- 9. Allow 'auto' priority source (priority derived from the task's timeline
--    position; recomputed on change). Upgrades an existing database.
ALTER TABLE tasks DROP CONSTRAINT IF EXISTS tasks_priority_source_check;
ALTER TABLE tasks ADD CONSTRAINT tasks_priority_source_check
    CHECK (priority_source IN ('user', 'ai', 'auto'));
